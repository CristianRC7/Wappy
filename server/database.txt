-- ======================================
-- TABLA: roles
-- ======================================
CREATE TABLE roles (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(50) UNIQUE NOT NULL,
    descripcion TEXT
);


-- ======================================
-- TABLA: usuarios
-- ======================================
CREATE TABLE usuarios (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    apellido VARCHAR(150) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    usuario VARCHAR(150) NOT NULL,
    contrasena TEXT NOT NULL,
    telefono VARCHAR(50),
    role_id INT REFERENCES roles(id)
);

-- ======================================
-- TABLA: grupos
-- ======================================
CREATE TABLE grupos (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    descripcion TEXT
);

-- ======================================
-- TABLA: grupo_usuario
-- Relación de usuarios con grupos
-- ======================================
CREATE TABLE grupo_usuario (
    id SERIAL PRIMARY KEY,
    usuario_id INT REFERENCES usuarios(id) ON DELETE CASCADE,
    grupo_id INT REFERENCES grupos(id) ON DELETE CASCADE,
    UNIQUE (usuario_id, grupo_id)
);

-- ======================================
-- TABLA: whatsapp_sesion
-- Sesión de WhatsApp asociada a un grupo
-- ======================================
CREATE TABLE whatsapp_sesion (
    id SERIAL PRIMARY KEY,
    grupo_id INT UNIQUE REFERENCES grupos(id) ON DELETE CASCADE,
    ruta_auth VARCHAR(255) NOT NULL
);

-- ======================================
-- TABLA: alertas
-- Alertas/Recordatorios del sistema
-- ======================================
CREATE TABLE alertas (
    id SERIAL PRIMARY KEY,
    titulo VARCHAR(200) NOT NULL,
    descripcion TEXT NOT NULL,
    fecha_hora TIMESTAMP NOT NULL,
    grupo_id INT REFERENCES grupos(id) ON DELETE CASCADE,
    creado_por INT REFERENCES usuarios(id) ON DELETE SET NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    estado VARCHAR(20) DEFAULT 'pendiente' -- pendiente, enviada, cancelada
);

-- ======================================
-- TABLA: alerta_destinatarios
-- Destinatarios de cada alerta
-- ======================================
CREATE TABLE alerta_destinatarios (
    id SERIAL PRIMARY KEY,
    alerta_id INT REFERENCES alertas(id) ON DELETE CASCADE,
    usuario_id INT REFERENCES usuarios(id) ON DELETE CASCADE,
    enviado BOOLEAN DEFAULT FALSE,
    fecha_envio TIMESTAMP,
    UNIQUE (alerta_id, usuario_id)
);
